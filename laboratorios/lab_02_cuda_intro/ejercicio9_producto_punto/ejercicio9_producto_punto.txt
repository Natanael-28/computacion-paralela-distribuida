#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <cuda_runtime.h>

#define N 4096
#define THREADS 256

__global__ void productoPunto(float *d_A, float *d_B, float *d_parciales, int n) {
    extern __shared__ float s_datos[];

    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;

    s_datos[tid] = (idx < n) ? d_A[idx] * d_B[idx] : 0.0f;
    __syncthreads();

    for (int stride = blockDim.x / 2; stride > 0; stride /= 2) {
        if (tid < stride)
            s_datos[tid] += s_datos[tid + stride];
        __syncthreads();
    }

    if (tid == 0) d_parciales[blockIdx.x] = s_datos[0];
}

float ejecutar(float *h_A, float *h_B, int numBloques) {
    float *d_A, *d_B, *d_parciales;
    float *h_parciales = (float*)malloc(numBloques * sizeof(float));

    cudaMalloc((void**)&d_A, N * sizeof(float));
    cudaMalloc((void**)&d_B, N * sizeof(float));
    cudaMalloc((void**)&d_parciales, numBloques * sizeof(float));

    cudaMemcpy(d_A, h_A, N * sizeof(float), cudaMemcpyHostToDevice);
    cudaMemcpy(d_B, h_B, N * sizeof(float), cudaMemcpyHostToDevice);

    int sharedBytes = THREADS * sizeof(float);
    productoPunto<<<numBloques, THREADS, sharedBytes>>>(d_A, d_B, d_parciales, N);
    cudaDeviceSynchronize();

    cudaMemcpy(h_parciales, d_parciales, numBloques * sizeof(float), cudaMemcpyDeviceToHost);

    float resultado = 0.0f;
    for (int i = 0; i < numBloques; i++) resultado += h_parciales[i];

    cudaFree(d_A); cudaFree(d_B); cudaFree(d_parciales);
    free(h_parciales);
    return resultado;
}

int main() {
    int numBloques = (N + THREADS - 1) / THREADS;

    float *h_A = (float*)malloc(N * sizeof(float));
    float *h_B = (float*)malloc(N * sizeof(float));

    printf("--- Caso 1: vectores de unos ---\n");
    for (int i = 0; i < N; i++) { h_A[i] = 1.0f; h_B[i] = 1.0f; }
    float r_gpu = ejecutar(h_A, h_B, numBloques);
    printf("Producto punto GPU = %.2f\n", r_gpu);
    printf("Resultado esperado = %d\n", N);
    printf("%s\n", (r_gpu == (float)N) ? "[OK]" : "[ERROR]");

    printf("\n--- Caso 2: vectores aleatorios ---\n");
    srand(42);
    for (int i = 0; i < N; i++) {
        h_A[i] = (float)rand() / RAND_MAX;
        h_B[i] = (float)rand() / RAND_MAX;
    }

    float r_cpu = 0.0f;
    for (int i = 0; i < N; i++) r_cpu += h_A[i] * h_B[i];

    r_gpu = ejecutar(h_A, h_B, numBloques);
    printf("Producto punto CPU = %.4f\n", r_cpu);
    printf("Producto punto GPU = %.4f\n", r_gpu);
    printf("Diferencia absoluta = %.6f\n", fabsf(r_cpu - r_gpu));
    printf("%s\n", (fabsf(r_cpu - r_gpu) < 1e-2f) ? "[OK] Coinciden dentro de la tolerancia" : "[ERROR]");

    free(h_A); free(h_B);
    return 0;
}
